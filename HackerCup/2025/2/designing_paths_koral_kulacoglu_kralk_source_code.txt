const int MOD = 1000000007;
const char nl = '\n';
const int MX = 100001; 
 
void solve() {
    int n, k, m;
    cin >> n >> k >> m;
    vector<vi> routes(m);
    vi processed(m, -1);
    vector<vpi> occ(n + 1);

    FOR(i, 0, m) {
        int L;
        cin >> L;
        routes[i].resize(L);
        FOR(j, 0, L) {
            cin >> routes[i][j];
            occ[routes[i][j]].push_back({i, j});
        }
    }

    const int INF = 1e9;
    vi dist(n + 1, INF);
    queue<int> que;

    dist[1] = 0;
    que.push(1);

    while (!que.empty()) {
        int v = que.front();
        que.pop();
        int d = dist[v];

        trav(o, occ[v]) {
            int rid = o.fir;
            int pos = o.sec;

            vi &route = routes[rid];
            int &pr = processed[rid];
            int s = sz(route);

            int l = max(pr + 1, pos + 1);
            int r = min(s - 1, pos + k);

            if (l <= r) {
                FOR(idx, l, r + 1) {
                    int u = route[idx];
                    if (dist[u] == INF) {
                        dist[u] = d + 1;
                        que.push(u);
                    }
                }
                pr = r;
            }
        }
    }

    ll ans = 0;
    FOR(i, 1, n + 1) {
        ll d = (dist[i] == INF ? -1LL : (ll)dist[i]);
        ans += d * (ll)i;
    }

    cout << ans << nl;
}

 
int main() {
    cin.tie(0)->sync_with_stdio(0); 
    cin.exceptions(cin.failbit);
 
    int T = 1;
	cin >> T;
	FOR (i, 1, T+1) {
		cout << "Case #" << i << ": ";
        solve();
    }
 
	return 0;
}
